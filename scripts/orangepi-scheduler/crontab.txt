# ============================================================================
# Portfolio Daily Recap — Market Session Scheduler
# Orange Pi 5 cron configuration (all times in UTC)
# ============================================================================
# Two entries per session to handle EU/US DST mismatch (~3 weeks/year).
# The dispatch script has built-in dedup, so double triggers are harmless.
# ============================================================================

# ── European Market Open ─────────────────────────────────────────────────────
# London open: 08:00 local → trigger at 08:02 local
# Summer (BST): 08:02 BST = 07:02 UTC
# Winter (GMT): 08:02 GMT = 08:02 UTC
2 7 * * 1-5   /opt/portfolio-dispatch/dispatch.sh eu_open  >> /opt/portfolio-dispatch/logs/cron.log 2>&1
2 8 * * 1-5   /opt/portfolio-dispatch/dispatch.sh eu_open  >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# ── Interactive Community Poll ───────────────────────────────────────────────
# 12:30 local (CET/CEST) → 10:30 UTC (summer) / 11:30 UTC (winter)
30 10 * * 1-5  /opt/portfolio-dispatch/dispatch.sh community_poll >> /opt/portfolio-dispatch/logs/cron.log 2>&1
30 11 * * 1-5  /opt/portfolio-dispatch/dispatch.sh community_poll >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# ── Stock Focus (Midday) ───────────────────────────────────────────────────────
# 14:00 local (CET/CEST) → 12:00 UTC (summer) / 13:00 UTC (winter)
0 12 * * 1-5   /opt/portfolio-dispatch/dispatch.sh stock_focus >> /opt/portfolio-dispatch/logs/cron.log 2>&1
0 13 * * 1-5   /opt/portfolio-dispatch/dispatch.sh stock_focus >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# ── U.S. Market Open ─────────────────────────────────────────────────────────
# NYSE open: 09:30 local → trigger at 09:32 local
# Summer (EDT): 09:32 EDT = 13:32 UTC
# Winter (EST): 09:32 EST = 14:32 UTC
32 13 * * 1-5  /opt/portfolio-dispatch/dispatch.sh us_open  >> /opt/portfolio-dispatch/logs/cron.log 2>&1
32 14 * * 1-5  /opt/portfolio-dispatch/dispatch.sh us_open  >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# ── Daily Crypto Pulse & Sentiment Recap ─────────────────────────────────────
# 18:00 local (CET/CEST) → 16:00 UTC (summer) / 17:00 UTC (winter)
0 16 * * *     /opt/portfolio-dispatch/dispatch.sh crypto_recap >> /opt/portfolio-dispatch/logs/cron.log 2>&1
0 17 * * *     /opt/portfolio-dispatch/dispatch.sh crypto_recap >> /opt/portfolio-dispatch/logs/cron.log 2>&1


# ── U.S. Market Close ────────────────────────────────────────────────────────
# NYSE close: 16:00 local → trigger at 16:02 local
# Summer (EDT): 16:02 EDT = 20:02 UTC
# Winter (EST): 16:02 EST = 21:02 UTC
2 20 * * 1-5   /opt/portfolio-dispatch/dispatch.sh us_close >> /opt/portfolio-dispatch/logs/cron.log 2>&1
2 21 * * 1-5   /opt/portfolio-dispatch/dispatch.sh us_close >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# ── DELAYED ENGAGEMENT (+60 min post-publish feed revival) ───────────────────
# Follow-up EU Open (+60m):        08:02 / 09:02 UTC
2 8 * * 1-5    /opt/portfolio-dispatch/dispatch.sh eu_open_followup >> /opt/portfolio-dispatch/logs/cron.log 2>&1
2 9 * * 1-5    /opt/portfolio-dispatch/dispatch.sh eu_open_followup >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# Follow-up Stock Focus (+60m):    13:00 / 14:00 UTC
0 13 * * 1-5   /opt/portfolio-dispatch/dispatch.sh stock_focus_followup >> /opt/portfolio-dispatch/logs/cron.log 2>&1
0 14 * * 1-5   /opt/portfolio-dispatch/dispatch.sh stock_focus_followup >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# Follow-up US Open (+60m):        14:32 / 15:32 UTC
32 14 * * 1-5  /opt/portfolio-dispatch/dispatch.sh us_open_followup >> /opt/portfolio-dispatch/logs/cron.log 2>&1
32 15 * * 1-5  /opt/portfolio-dispatch/dispatch.sh us_open_followup >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# Follow-up US Close (+60m):       21:02 / 22:02 UTC
2 21 * * 1-5   /opt/portfolio-dispatch/dispatch.sh us_close_followup >> /opt/portfolio-dispatch/logs/cron.log 2>&1
2 22 * * 1-5   /opt/portfolio-dispatch/dispatch.sh us_close_followup >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# ── Weekly Recap (Saturday) ──────────────────────────────────────────────────
# 10:00 CET = 09:00 UTC (summer: 08:00 UTC)
0 8 * * 6     /opt/portfolio-dispatch/dispatch.sh weekly_sat >> /opt/portfolio-dispatch/logs/cron.log 2>&1
0 9 * * 6     /opt/portfolio-dispatch/dispatch.sh weekly_sat >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# ── Weekly Recap (Sunday) ────────────────────────────────────────────────────
# 22:00 CET = 21:00 UTC (summer: 20:00 UTC)
0 20 * * 0    /opt/portfolio-dispatch/dispatch.sh weekly_sun >> /opt/portfolio-dispatch/logs/cron.log 2>&1
0 21 * * 0    /opt/portfolio-dispatch/dispatch.sh weekly_sun >> /opt/portfolio-dispatch/logs/cron.log 2>&1
