# ============================================================================
# Portfolio Daily Recap — Market Session Scheduler
# Orange Pi 5 cron configuration (Timezone: Europe/Rome)
# ============================================================================
# Configured in Italian local time (Europe/Rome - CET/CEST).
# Market opening and closing times are strictly enforced by dispatch.sh
# guards (checking live New York and Rome exchange times) to eliminate any
# risk of premature execution during daylight saving time (DST) shifts.
# ============================================================================
CRON_TZ=Europe/Rome

# ── European Market Open (Mon-Fri 09:02 ITA) ──────────────────────────────────
# Milan/Frankfurt open at 09:00 local; London opens at 08:00 GMT/BST (09:00 ITA)
2 9 * * 1-5       /opt/portfolio-dispatch/dispatch.sh eu_open >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# ── Dividend Announcement Check (Mon & Thu 09:45 ITA) ─────────────────────────
45 9 * * 1,4      /opt/portfolio-dispatch/dispatch.sh dividend_post >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# ── Event Monitor (Earnings & Dividend Pay Days) ──────────────────────────────
# Window 1 (Mattina Europea - Mon-Fri 10:00 ITA):
#   Prior-day AMC (post-market close) earnings after full call digestion & European morning releases
0 10 * * 1-5      /opt/portfolio-dispatch/dispatch.sh event_monitor >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# Window 2 (Poco dopo apertura USA - Mon-Fri):
#   BMO (pre-market open) earnings once regular trading session is established (~10:15 ET)
#   - Normal (summer EDT & winter EST): 16:15 ITA
#   - Transition weeks (autumn/spring mismatch): 15:15 ITA
#   dispatch.sh verifies New York time >= 10:00 ET and auto-deduplicates.
15 15,16 * * 1-5  /opt/portfolio-dispatch/dispatch.sh event_monitor >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# ── Stock News & Catalyst Follow-up Check - Morning (Daily 11:30 ITA) ─────────
30 11 * * *       /opt/portfolio-dispatch/dispatch.sh stock_news >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# ── Interactive Community Poll (Mon, Wed, Fri 12:30 ITA) ──────────────────────
# Mon: Macro Calendar | Wed: Thematic Strategy | Fri: Weekly Top Performers
30 12 * * 1,3,5   /opt/portfolio-dispatch/dispatch.sh community_poll >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# ── Copy Trading Education & Megatrend Deep-Dive ──────────────────────────────
# Mon-Fri: 13:30 ITA | Saturday: 12:00 ITA
30 13 * * 1-5     /opt/portfolio-dispatch/dispatch.sh copy_trading >> /opt/portfolio-dispatch/logs/cron.log 2>&1
0 12 * * 6        /opt/portfolio-dispatch/dispatch.sh copy_trading >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# ── Stock Focus (Mon-Fri 14:00 ITA) ───────────────────────────────────────────
0 14 * * 1-5      /opt/portfolio-dispatch/dispatch.sh stock_focus >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# ── U.S. Market Open (Mon-Fri) ────────────────────────────────────────────────
# Wall Street opens at 09:30 America/New_York:
# - Normal (summer EDT & winter EST): 15:30 ITA -> trigger at 15:32 ITA
# - Transition weeks (autumn/spring mismatch): 14:30 ITA -> trigger at 14:32 ITA
# dispatch.sh verifies New York time >= 09:30 ET and auto-deduplicates.
32 14,15 * * 1-5  /opt/portfolio-dispatch/dispatch.sh us_open >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# ── Stock News & Catalyst Follow-up Check - Afternoon (Daily 17:00 ITA) ───────
0 17 * * *        /opt/portfolio-dispatch/dispatch.sh stock_news >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# ── Daily Crypto Pulse & Sentiment Recap ──────────────────────────────────────
# Mon-Sat: 18:00 ITA | Sunday: 14:00 ITA
0 18 * * 1-6      /opt/portfolio-dispatch/dispatch.sh crypto_recap >> /opt/portfolio-dispatch/logs/cron.log 2>&1
0 14 * * 0        /opt/portfolio-dispatch/dispatch.sh crypto_recap >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# ── U.S. Market Close (Mon-Fri) ───────────────────────────────────────────────
# Wall Street closes at 16:00 America/New_York:
# - Normal (summer EDT & winter EST): 22:00 ITA -> trigger at 22:02 ITA
# - Transition weeks (autumn/spring mismatch): 21:00 ITA -> trigger at 21:02 ITA
# dispatch.sh verifies New York time >= 16:00 ET and auto-deduplicates.
2 21,22 * * 1-5   /opt/portfolio-dispatch/dispatch.sh us_close >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# ── Monthly Market & Portfolio Recap (Last calendar day of each month) ────────
# Automatically replaces normal US close on the final calendar day of each month
2 21,22 28-31 * * [ "$(date -u -d '+1 day' +%d)" = "01" ] && /opt/portfolio-dispatch/dispatch.sh monthly >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# ── Stock News & Catalyst Follow-up Check - Evening (Daily 23:30 ITA) ─────────
30 23 * * *       /opt/portfolio-dispatch/dispatch.sh stock_news >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# ── Weekly Recap (Saturday 10:00 ITA) ─────────────────────────────────────────
0 10 * * 6        /opt/portfolio-dispatch/dispatch.sh weekly_sat >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# ── Sunday: Weekly Macro & Portfolio Outlook (Sunday 22:00 ITA) ───────────────
0 22 * * 0        /opt/portfolio-dispatch/dispatch.sh weekly_sun >> /opt/portfolio-dispatch/logs/cron.log 2>&1

# ── Nightly Dashboard & Metrics Sync (Daily 02:00 ITA) ────────────────────────
0 2 * * *         /opt/portfolio-dispatch/dispatch.sh dashboard_sync >> /opt/portfolio-dispatch/logs/cron.log 2>&1
